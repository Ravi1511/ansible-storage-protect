#!/bin/bash

echo "===== GPFS Setup Started ====="

# =========================================
# STEP 1: Verify GPFS
# =========================================
echo "Checking GPFS state..."
mmgetstate -a

# =========================================
# STEP 2: Create NSD file
# =========================================
echo "Creating nsd.txt..."

cat <<EOF > nsd.txt
%nsd:
  device=/dev/dm-7
  nsd=nsd1
  servers=$(hostname -f)
  usage=dataAndMetadata

%nsd:
  device=/dev/dm-8
  nsd=nsd2
  servers=$(hostname -f)
  usage=dataAndMetadata

%nsd:
  device=/dev/dm-9
  nsd=nsd3
  servers=$(hostname -f)
  usage=dataAndMetadata

%nsd:
  device=/dev/dm-10
  nsd=nsd4
  servers=$(hostname -f)
  usage=dataAndMetadata

%nsd:
  device=/dev/dm-11
  nsd=nsd5
  servers=$(hostname -f)
  usage=dataAndMetadata
EOF


# =========================================
# STEP 3: Cleanup old setup (safe)
# =========================================
echo "Cleaning old GPFS configuration..."

mmumount all -a 2>/dev/null

for fs in gpfs_main gpfs1 gpfs1nsd gpfs2nsd gpfs3nsd gpfs4nsd gpfs5nsd
do
    mmdelfs $fs 2>/dev/null
done

for nsd in gpfs1nsd gpfs2nsd gpfs3nsd gpfs4nsd gpfs5nsd nsd1 nsd2 nsd3 nsd4 nsd5
do
    mmdelnsd $nsd 2>/dev/null
done


# =========================================
# STEP 4: Create NSDs
# =========================================
echo "Creating NSDs..."
mmcrnsd -F nsd.txt

if [ $? -ne 0 ]; then
    echo "❌ NSD creation failed"
    exit 1
fi

# =========================================
# STEP 5: Create disk list file
# =========================================
echo "Creating fs_disks.txt..."

cat <<EOF > fs_disks.txt
nsd1
nsd2
nsd3
nsd4
nsd5
EOF


# =========================================
# STEP 6: Create filesystem
# =========================================
echo "Creating filesystem gpfs_main..."

mmcrfs gpfs_main -F fs_disks.txt -T /gpfs_main -A yes

if [ $? -ne 0 ]; then
    echo "❌ Filesystem creation failed"
    exit 1
fi


# =========================================
# STEP 7: Mount filesystem
# =========================================
echo "Mounting filesystem..."

mmmount gpfs_main -a

# Verify mount
df -h | grep gpfs_main


# =========================================
# STEP 8: Create filesets
# =========================================
echo "Creating filesets..."

mmcrfileset gpfs_main testfs
mmcrfileset gpfs_main fileset_1 --inode-space new
mmcrfileset gpfs_main fileset_2 --inode-space new
mmcrfileset gpfs_main fileset_3 --inode-space new


# =========================================
# STEP 9: Link filesets
# =========================================
echo "Linking filesets..."

mmlinkfileset gpfs_main fileset_1 -J /gpfs_main/fileset_1
mmlinkfileset gpfs_main fileset_2 -J /gpfs_main/fileset_2
mmlinkfileset gpfs_main fileset_3 -J /gpfs_main/fileset_3
mmlinkfileset gpfs_main testfs -J /gpfs_main/testfs


# =========================================
# STEP 10: Verify
# =========================================
echo "Final verification..."

mmlsfileset gpfs_main
mmdf gpfs_main

echo "===== GPFS Setup Completed Successfully ====="
