Step 3: Register Nodes on SP Servers
On PETASCALE-SP01 (9.11.53.28):
# SSH to sp-server-01
ssh root@9.11.53.28

# Register node for hsm-client-03
dsmadmc -id=admin -password=<admin_password>

# Inside dsmadmc:
register node hsm-client-03-sp01 <node_password> domain=STANDARD maxnummp=4
grant authority hsm-client-03-sp01 classes=client
quit

On PETASCALE-SP03 (9.11.53.254):
# SSH to sp-server-03
ssh root@9.11.53.254

# Register node for hsm-client-03
dsmadmc -id=admin -password=<admin_password>

# Inside dsmadmc:
register node hsm-client-03-sp03 <node_password> domain=STANDARD maxnummp=4
grant authority hsm-client-03-sp03 classes=client
quit

Step 4: Test Connectivity from HSM Client
# On hsm-client-03
cd /opt/tivoli/tsm/client/hsm/bin

# Test connection to PETASCALE-SP01
./dsmc query session -servername=PETASCALE-SP01
# Enter password when prompted: <node_password>

# Test connection to PETASCALE-SP03
./dsmc query session -servername=PETASCALE-SP03
# Enter password when prompted: <node_password>

Step 5: Test Backup Operations
# On hsm-client-03

# Create test files in different filesets
mkdir -p /gpfs_main/fileset_2/test
mkdir -p /gpfs_main/fileset_3/test
echo "Test data for SP01" > /gpfs_main/fileset_2/test/file1.txt
echo "Test data for SP03" > /gpfs_main/fileset_3/test/file1.txt

# Backup fileset_2 to PETASCALE-SP01 (default server)
./dsmc incremental /gpfs_main/fileset_2/test/

# Backup fileset_3 to PETASCALE-SP03 (explicit server)
./dsmc incremental /gpfs_main/fileset_3/test/ -servername=PETASCALE-SP03

# Query backed up files on each server
./dsmc query backup "/gpfs_main/fileset_2/*" -servername=PETASCALE-SP01
./dsmc query backup "/gpfs_main/fileset_3/*" -servername=PETASCALE-SP03

Troubleshooting:
If playbook fails:
# Check syntax
ansible-playbook playbooks/petascale_configure.yml --syntax-check

# Run with verbose output
ansible-playbook playbooks/petascale_configure.yml \
  -i playbooks/inventory/petascale.ini \
  --limit hsm-client-03 \
  -e "configure_clients=true" \
  -vvv

If connection fails:
# Check network connectivity
ping 9.11.53.28
ping 9.11.53.254

# Check SP Server is listening
telnet 9.11.53.28 1500
telnet 9.11.53.254 1500

# Check dsm.sys syntax
./dsmc query session -servername=PETASCALE-SP01 -optfile=/opt/tivoli/tsm/client/hsm/bin/dsm.opt
